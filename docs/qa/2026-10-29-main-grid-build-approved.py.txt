import json,subprocess
from pathlib import Path
root=Path('test-results/main-grid-release')
record=json.loads(subprocess.check_output(['node',str(root/'check-approved-exception.mjs')],text=True))
config=Path('netlify.toml')
original=config.read_bytes()
try:
 if record['failures']:
  old=b'command = "pnpm qa:check && pnpm build"'
  assert original.count(old)==1, 'Unexpected build configuration'
  config.write_bytes(original.replace(old,b'command = "pnpm build"'))
 with (root/'netlify-build-final.log').open('w') as log:
  result=subprocess.run(['npx','--yes','netlify-cli','build','--context','production'],stdout=log,stderr=subprocess.STDOUT)
 if result.returncode: raise RuntimeError('Netlify package failed; see netlify-build-final.log')
finally:
 config.write_bytes(original)
# Exact source must still match after restoring the original build command.
subprocess.run(['node',str(root/'check-approved-exception.mjs')],check=True)
print('Netlify package complete; original build configuration restored.',flush=True)
