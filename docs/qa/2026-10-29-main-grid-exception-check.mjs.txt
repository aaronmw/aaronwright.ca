import { readFileSync, readdirSync, writeFileSync } from 'node:fs'
import { compare, completeness, sourceHash, summarize } from '../../scripts/qa/report.mjs'
const reportPath = 'qa/results/' + readdirSync('qa/results').filter(f => f.endsWith('.json')).sort().at(-1)
const report = JSON.parse(readFileSync(reportPath))
const baseline = JSON.parse(readFileSync('qa/baseline.json'))
const incomplete = completeness(report)
if (incomplete.length) throw new Error(incomplete.join('\n'))
if (report.sourceHash !== sourceHash()) throw new Error('Source changed after verification')
if (baseline.id !== '2026-10-30T01-01-36-132Z') throw new Error('Accepted baseline changed')
const failures = compare(report, baseline)
const row = summarize(report.samples).find(r => r.profile === 'small-laptop-slow' && r.task === 'close-viewer')
const approved = failures.length === 1 && failures[0].startsWith('small-laptop-slow/close-viewer: frameGapP95Ms ') && row.frameGapP95Ms <= 50.1 + 1e-6
if (failures.length && !approved) throw new Error('Outside the approved exception:\n' + failures.join('\n'))
const record = {
  reportPath, sourceHash: report.sourceHash, baseline: baseline.id,
  disposition: failures.length ? 'approved one-release exception' : 'normal gate passed; exception unused',
  authorization: 'Aaron: Allow this isolated exception. Subsequently authorized inclusion of the current video updates.',
  failures, checkedAt: new Date().toISOString(),
}
writeFileSync('test-results/main-grid-release/approved-gate.json', JSON.stringify(record, null, 2))
console.log(JSON.stringify(record))
